        train/.style={block, fill=orange!10, minimum width=3.0cm, font=\scriptsize\sffamily\bfseries},

    % Main acoustic and spike front end.
    \node[source] (target) at (0,0) {Target state\\$(r,\theta,\psi)$};
    \node[block, right=0.65cm of target] (call) {FM call\\+ corollary\\discharge};
    \node[block, right=0.65cm of call] (echo) {Echo simulator\\delay, attenuation,\\noise, spectral cue};
    \node[block, right=0.65cm of echo] (cochlea) {Cochlear front end\\IIR filterbank\\+ LIF encoder};

    % Parallel pathway populations.
    \node[pathway, right=0.95cm of cochlea, yshift=1.25cm] (dist) {Distance pathway\\call-copy--echo coincidence\\$P_r\in\mathbb{R}^{180}$};
    \node[pathway, right=0.95cm of cochlea] (az) {Azimuth pathway\\ITD + ILD populations\\$P_\theta^{ITD},P_\theta^{ILD}\in\mathbb{R}^{91}$};
    \node[pathway, right=0.95cm of cochlea, yshift=-1.25cm] (el) {Elevation pathway\\spectral templates\\$P_\psi\in\mathbb{R}^{91}$};

    % Fixed readouts and feature construction.
    \node[readout, right=0.95cm of az, yshift=0.65cm] (fixed) {Fixed readouts\\COM, calibration,\\SC/CANN};
    \node[readout, right=0.95cm of az, yshift=-0.65cm] (features) {Feature vector $z$\\raw populations\\+ $3$ readouts + $18$ confidence};

    % Trainable fusion.
    \node[train, right=0.95cm of fixed, yshift=-0.3cm] (snn) {Residual SNN\\Linear--LIF--Linear--LIF\\$474\rightarrow96\rightarrow96\rightarrow5$};
    \node[block, right=0.75cm of snn] (output) {Final estimate\\$\hat{r},\hat{\theta},\hat{\psi}$};

    % Main arrows.
    \draw[arrow] (target) -- (call);
    \draw[arrow] (call) -- node[above, note] {waveform} (echo);
    \draw[arrow] (echo) -- node[above, note] {binaural\\waveform} (cochlea);
    \draw[arrow] ([yshift=0.12cm]cochlea.east) -- ++(0.35,0) |- node[pos=0.25, above, note] {CSR} (dist.west);

    \draw[arrow] ([yshift=-0.12cm]cochlea.east) -- ++(0.35,0) |- node[pos=0.25, below, note] {CSR} (el.west);

    \draw[dataarrow] (call.north) |- ++(1.2,0.85) -| node[pos=0.32, above, note] {internal timing reference} (dist.north);

    \draw[arrow] (dist.east) -- ++(0.25,0) |- (fixed.west);
